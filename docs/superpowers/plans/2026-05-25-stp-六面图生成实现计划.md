# STP 文件六面图生成实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将产品库的 STP 文件上传改为异步生成六面图功能，用户上传 STP 文件后，后端使用 Python + OpenCASCADE 渲染 6 张产品视图，自动添加到参考图。

**Architecture:** 采用 Redis 队列的异步处理架构：Go 后端接收 STP 文件上传 → 写入 Redis 队列 → Python Worker 消费任务渲染六面图 → 结果更新到产品数据库。前端轮询渲染状态。

**Tech Stack:** Go (gin), GORM, Redis, 阿里云 OSS, Python 3.10+, OpenCASCADE (OCCT)

package handler

import (
	"fmt"
	"net/http"
	"os"
	"path/filepath"
	"strings"
	"time"

	"advertising-image/backend/internal/model"
	"advertising-image/backend/internal/storage"
	"advertising-image/backend/internal/cache"

	"github.com/gin-gonic/gin"
	"github.com/google/uuid"
)

// UploadStpFile 上传 STP 文件并触发渲染
func UploadStpFile(c *gin.Context) {
	productID := c.Param("id")

	// 解析 product ID
	var id uint
	if _, err := fmt.Sscanf(productID, "%d", &id); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "无效的产品ID"})
		return
	}

	// 查找产品
	var product model.ProductPrompt
	if err := model.DB.First(&product, id).Error; err != nil {
		c.JSON(http.StatusNotFound, gin.H{"error": "产品不存在"})
		return
	}

	// 获取当前用户
	userID := c.GetUint("user_id")

	// 接收文件
	file, err := c.FormFile("file")
	if err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "请上传 STP 文件"})
		return
	}

	// 校验文件类型
	ext := strings.ToLower(filepath.Ext(file.Filename))
	if ext != ".stp" && ext != ".step" {
		c.JSON(http.StatusBadRequest, gin.H{"error": "仅支持 .stp 和 .step 文件"})
		return
	}

	// 校验文件大小 (< 50MB)
	if file.Size > 50*1024*1024 {
		c.JSON(http.StatusBadRequest, gin.H{"error": "文件大小不能超过 50 MB"})
		return
	}

	// 保存到本地临时目录
	tmpDir := "./uploads/stp"
	os.MkdirAll(tmpDir, 0755)
	tmpPath := filepath.Join(tmpDir, fmt.Sprintf("%d_%d%s", userID, time.Now().UnixNano(), ext))

	if err := c.SaveUploadedFile(file, tmpPath); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "保存文件失败"})
		return
	}

	// 上传到 OSS
	ossDir := "products/stp"
	url, err := storage.MoveToOSS(tmpPath, ossDir)
	if err != nil {
		os.Remove(tmpPath)
		c.JSON(http.StatusInternalServerError, gin.H{"error": "上传到 OSS 失败"})
		return
	}

	// 生成任务 ID
	taskID := uuid.New().String()

	// 写入 Redis Hash（任务状态）
	if cache.RDB != nil {
		ctx := c.Request.Context()
		taskKey := fmt.Sprintf("stp_render_task:%s", taskID)
		cache.RDB.HSet(ctx, taskKey, map[string]interface{}{
			"task_id":      taskID,
			"product_id":   id,
			"stp_file_url": url,
			"user_id":      userID,
			"status":       "pending",
			"created_at":   time.Now().Format(time.RFC3339),
		})
		// 设置 24 小时过期
		cache.RDB.Expire(ctx, taskKey, 24*time.Hour)

		// 写入队列
		cache.RDB.LPush(ctx, "stp_render_queue", taskID)
	}

	// 更新产品数据库
	model.DB.Model(&product).Updates(map[string]interface{}{
		"stp_file_url":  url,
		"render_status": "processing",
		"render_error":  "",
	})

	c.JSON(http.StatusOK, gin.H{
		"code": 0,
		"data": gin.H{
			"task_id":      taskID,
			"stp_file_url": url,
			"render_status": "processing",
		},
	})
}

// GetRenderStatus 查询渲染状态
func GetRenderStatus(c *gin.Context) {
	productID := c.Param("id")

	var id uint
	if _, err := fmt.Sscanf(productID, "%d", &id); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "无效的产品ID"})
		return
	}

	var product model.ProductPrompt
	if err := model.DB.First(&product, id).Error; err != nil {
		c.JSON(http.StatusNotFound, gin.H{"error": "产品不存在"})
		return
	}

	// 从 reference_images 中筛选六面图
	sixViews := make([]model.ReferenceImage, 0)
	for _, img := range product.ReferenceImages {
		if strings.HasPrefix(img.Description, "[六面-") {
			sixViews = append(sixViews, img)
		}
	}

	c.JSON(http.StatusOK, gin.H{
		"code": 0,
		"data": gin.H{
			"stp_file_url":  product.StpFileURL,
			"render_status": product.RenderStatus,
			"render_error":  product.RenderError,
			"six_views":     sixViews,
		},
	})
}

// UpdateRenderResult Python Worker 调用此接口更新渲染结果
func UpdateRenderResult(c *gin.Context) {
	var req struct {
		TaskID     string                   `json:"task_id"`
		Status     string                   `json:"status"` // completed / failed
		SixViews   []model.ReferenceImage   `json:"six_views"`
		ErrorMsg   string                   `json:"error_message"`
	}

	if err := c.ShouldBindJSON(&req); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "请求格式错误"})
		return
	}

	// 从 Redis 获取任务信息
	if cache.RDB == nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "Redis 未配置"})
		return
	}

	ctx := c.Request.Context()
	taskKey := fmt.Sprintf("stp_render_task:%s", req.TaskID)
	productIDStr, err := cache.RDB.HGet(ctx, taskKey, "product_id").Result()
	if err != nil {
		c.JSON(http.StatusNotFound, gin.H{"error": "任务不存在"})
		return
	}

	var productID uint
	fmt.Sscanf(productIDStr, "%d", &productID)

	// 更新任务状态
	cache.RDB.HSet(ctx, taskKey, map[string]interface{}{
		"status":       req.Status,
		"error_message": req.ErrorMsg,
		"completed_at": time.Now().Format(time.RFC3339),
	})

	// 更新产品数据库
	updates := map[string]interface{}{
		"render_status": req.Status,
		"render_error":  req.ErrorMsg,
	}

	if req.Status == "completed" && len(req.SixViews) > 0 {
		// 获取当前 reference_images，追加六面图
		var product model.ProductPrompt
		if err := model.DB.First(&product, productID).Error; err == nil {
			existingImgs := product.ReferenceImages
			// 移除旧六面图
			filtered := make([]model.ReferenceImage, 0)
			for _, img := range existingImgs {
				if !strings.HasPrefix(img.Description, "[六面-") {
					filtered = append(filtered, img)
				}
			}
			// 添加新六面图
			updates["reference_images"] = append(filtered, req.SixViews...)
		}
	}

	model.DB.Model(&model.ProductPrompt{}).Where("id = ?", productID).Updates(updates)

	c.JSON(http.StatusOK, gin.H{"code": 0, "message": "更新成功"})
}
```

// STP 文件上传
const stpInputRef = ref(null)
const stpUploading = ref(false)

const triggerStpUpload = () => {
  stpInputRef.value?.click()
}

const handleStpUpload = async (event) => {
  const file = event.target.files[0]
  if (!file) return

  // 校验文件类型
  const ext = file.name.split('.').pop().toLowerCase()
  if (!['stp', 'step'].includes(ext)) {
    Message.error('仅支持 .stp 和 .step 文件')
    return
  }

  // 校验文件大小
  if (file.size > 50 * 1024 * 1024) {
    Message.error('文件大小不能超过 50 MB')
    return
  }

  stpUploading.value = true
  try {
    const formData = new FormData()
    formData.append('file', file)

    const res = await axios.post(`/api/product-prompts/${formData.id}/stp`, formData, {
      headers: { 'Content-Type': 'multipart/form-data' }
    })

    if (res.data.code === 0) {
      formData.stp_file_url = res.data.data.stp_file_url
      formData.render_status = res.data.data.render_status
      Message.success('STP 文件上传成功，正在渲染六面图')
      startPollingRenderStatus()
    }
  } catch (err) {
    Message.error(err.response?.data?.error || '上传失败')
  } finally {
    stpUploading.value = false
    event.target.value = ''
  }
}

// 轮询渲染状态
let pollingTimer = null

const startPollingRenderStatus = () => {
  if (pollingTimer) clearInterval(pollingTimer)

  pollingTimer = setInterval(async () => {
    try {
      const res = await axios.get(`/api/product-prompts/${formData.id}/render-status`)
      if (res.data.code === 0) {
        formData.render_status = res.data.data.render_status
        formData.render_error = res.data.data.render_error

        if (res.data.data.render_status === 'completed') {
          // 更新 reference_images
          formData.reference_images = res.data.data.six_views
          clearInterval(pollingTimer)
          pollingTimer = null
          Message.success('六面图渲染完成')
        } else if (res.data.data.render_status === 'failed') {
          clearInterval(pollingTimer)
          pollingTimer = null
          Message.error(res.data.data.render_error || '渲染失败')
        }
      }
    } catch (err) {
      console.error('轮询渲染状态失败', err)
    }
  }, 3000)
}

onUnmounted(() => {
  if (pollingTimer) clearInterval(pollingTimer)
})
```

import oss2
import io
from PIL import Image
from config import OSS_ACCESS_KEY, OSS_ACCESS_SECRET, OSS_ENDPOINT, OSS_BUCKET, OSS_DOMAIN

class OSSClient:
    def __init__(self):
        self.auth = oss2.Auth(OSS_ACCESS_KEY, OSS_ACCESS_SECRET)
        self.bucket = oss2.Bucket(self.auth, OSS_ENDPOINT, OSS_BUCKET)

    def upload_image(self, image: Image.Image, object_key: str) -> str:
        """上传 PIL Image 到 OSS"""
        buffer = io.BytesIO()
        image.save(buffer, format='PNG')
        buffer.seek(0)

        self.bucket.put_object(object_key, buffer)

        if OSS_DOMAIN:
            return f"{OSS_DOMAIN}/{object_key}"
        return f"https://{OSS_BUCKET}.{OSS_ENDPOINT}/{object_key}"

    def download_file(self, url: str, local_path: str) -> str:
        """下载 OSS 文件到本地"""
        self.bucket.get_object_to_file(url, local_path)
        return local_path

# 全局单例
oss_client = OSSClient() if OSS_ACCESS_KEY and OSS_ACCESS_SECRET else None

from OCP.BRepMesh_IncrementalMesh import BRepMesh_IncrementalMesh
from OCP.BRepBndLib import BRepBndLib_AddMesh
from OCP.Bnd import Bnd_Box
from OCP.BRepBuilderAPI import BRepBuilderAPI_MakeShape
from OCP.StlAPI import StlAPI_Reader
import tempfile
import os

class StepLoader:
    def __init__(self):
        pass

    def load(self, file_path: str) -> tuple[TopoDS_Shape, Bnd_Box]:
        """加载 STP 文件，返回 shape 和包围盒"""
        from OCP.IFSelect import IFSelect_RetDone
        from OCP.STEPControl import STEPControl_Reader

        reader = STEPControl_Reader()
        status = reader.ReadFile(file_path)

        if status != IFSelect_RetDone:
            raise ValueError(f"Failed to read STEP file: {file_path}")

        reader.TransferRoots()
        shape = reader.OneShape()

        # 计算包围盒
        bbox = Bnd_Box()
        BRepBndLib_AddMesh(shape, bbox)

        return shape, bbox

    def get_mesh_data(self, shape: TopoDS_Shape, bbox: Bnd_Box) -> tuple:
        """将 shape 三角化并返回顶点数据"""
        # 网格化
        mesh = BRepMesh_IncrementalMesh(shape, 0.1)  # deflection = 0.1mm
        mesh.Perform()

        # 获取三角网格数据
        # 返回顶点数组和面索引数组
        # 这需要更复杂的 OCCT API 调用，此处为简化示例
        min_p = bbox.CornerMin()
        max_p = bbox.CornerMax()

        # 包围盒尺寸
        dx = max_p.X() - min_p.X()
        dy = max_p.Y() - min_p.Y()
        dz = max_p.Z() - min_p.Z()

        return {
            'min': (min_p.X(), min_p.Y(), min_p.Z()),
            'max': (max_p.X(), max_p.Y(), max_p.Z()),
            'size': (dx, dy, dz)
        }
```

import numpy as np

# 6 个视角配置
VIEW_CONFIGS = [
    {'name': 'front',   'pos': (0, -1, 0),   'target': (0, 0, 0),  'up': (0, 0, 1)},
    {'name': 'back',    'pos': (0, 1, 0),    'target': (0, 0, 0),  'up': (0, 0, 1)},
    {'name': 'left',    'pos': (-1, 0, 0),   'target': (0, 0, 0),  'up': (0, 0, 1)},
    {'name': 'right',   'pos': (1, 0, 0),     'target': (0, 0, 0),  'up': (0, 0, 1)},
    {'name': 'top',     'pos': (0, 0, 1),     'target': (0, 0, 0),  'up': (0, -1, 0)},
    {'name': 'bottom',  'pos': (0, 0, -1),    'target': (0, 0, 0),  'up': (0, 1, 0)},
]

class Camera:
    def __init__(self, pos: tuple, target: tuple, up: tuple, distance: float):
        self.pos = np.array(pos) * distance
        self.target = np.array(target)
        self.up = np.array(up)
        # 确保 up 向量与视线方向正交
        forward = self.target - self.pos
        forward = forward / np.linalg.norm(forward)
        self.up = self.up - np.dot(self.up, forward) * forward
        self.up = self.up / np.linalg.norm(self.up)

    def get_view_matrix(self) -> np.ndarray:
        """计算 View Matrix (look-at matrix)"""
        f = self.target - self.pos
        f = f / np.linalg.norm(f)
        s = np.cross(f, self.up)
        s = s / np.linalg.norm(s)
        u = np.cross(s, f)

        view = np.identity(4)
        view[0, :3] = s
        view[1, :3] = u
        view[2, :3] = -f
        view[0, 3] = -np.dot(s, self.pos)
        view[1, 3] = -np.dot(u, self.pos)
        view[2, 3] = np.dot(f, self.pos)

        return view

    @staticmethod
    def calculate_distance(bbox_size: tuple, factor: float = 2.5) -> float:
        """根据包围盒计算相机距离"""
        max_dim = max(bbox_size)
        return max_dim * factor

def create_cameras_for_view(mesh_info: dict, factor: float = 2.5) -> list:
    """为 6 个视角创建相机"""
    dx, dy, dz = mesh_info['size']
    distance = max(dx, dy, dz) * factor

    cameras = []
    for config in VIEW_CONFIGS:
        cam = Camera(
            pos=config['pos'],
            target=config['target'],
            up=config['up'],
            distance=distance
        )
        cameras.append({
            'name': config['name'],
            'camera': cam,
            'description': f"[六面-{config['name']}]"
        })

    return cameras
```

import numpy as np
from config import RENDER_RESOLUTION, RENDER_BG_COLOR

class ImageRenderer:
    def __init__(self, resolution: int = RENDER_RESOLUTION):
        self.resolution = resolution
        self.bg_color = RENDER_BG_COLOR

    def create_view_image(self, shape, camera, view_name: str) -> Image.Image:
        """
        渲染单个视角

        注意：这里需要使用 OCCT 的 3D 渲染功能。
        由于 OCCT 的 Qt 依赖，在纯 CPU 无头环境下，
        我们使用简单的线框渲染或伪 3D 投影。

        实际生产环境中，建议使用 Open3D + OCC 的组合，
        或者使用 cadquery-freecad-module 的渲染功能。
        """
        # 创建白色背景图
        img = Image.new('RGB', (self.resolution, self.resolution), self.bg_color)
        draw = ImageDraw.Draw(img)

        # 获取包围盒
        # 在真实实现中，这里会进行实际的 3D 渲染
        # 由于 OCCT 的可视化依赖复杂，我们提供简化的实现
        center_x = self.resolution // 2
        center_y = self.resolution // 2

        # 绘制占位圆（实际会用真实 3D 渲染替换）
        # 这是为了让代码能运行，实际渲染需要 OCCT + Qt 环境
        radius = int(self.resolution * 0.3)
        draw.ellipse(
            [center_x - radius, center_y - radius, center_x + radius, center_y + radius],
            outline=(200, 200, 200),
            width=2
        )

        return img

    def save_image(self, image: Image.Image, output_path: str):
        """保存图像到文件"""
        image.save(output_path, format='PNG')

```python
#!/usr/bin/env python3
import os
import sys
import time
import tempfile
import requests
from redis import Redis
from config import REDIS_HOST, REDIS_PORT, REDIS_DB, GO_API_BASE_URL, RENDER_RESOLUTION

# 添加当前目录到路径
sys.path.insert(0, os.path.dirname(__file__))

from renderer.step_loader import StepLoader
from renderer.camera import create_cameras_for_view, VIEW_CONFIGS
from renderer.image_saver import ImageRenderer
from oss.client import oss_client, OSSClient

# 初始化组件
redis_client = Redis(host=REDIS_HOST, port=REDIS_PORT, db=REDIS_DB, decode_responses=True)
step_loader = StepLoader()
renderer = ImageRenderer()

QUEUE_NAME = "stp_render_queue"
MAX_RETRIES = 3

def download_stp_file(oss_url: str) -> str:
    """下载 STP 文件到本地临时目录"""
    tmp_dir = tempfile.mkdtemp(prefix="stp_")
    local_path = os.path.join(tmp_dir, "model.stp")

    if oss_client:
        oss_client.download_file(oss_url, local_path)
    else:
        # 如果没有 OSS 配置，直接从 URL 下载
        response = requests.get(oss_url, timeout=60)
        with open(local_path, 'wb') as f:
            f.write(response.content)

    return local_path

def render_six_views(stp_path: str) -> list:
    """渲染六面图，返回 6 张图像的 OSS URL"""
    # 加载 STP 文件
    shape, bbox = step_loader.load(stp_path)
    mesh_info = step_loader.get_mesh_data(shape, bbox)

    # 创建 6 个相机
    cameras = create_cameras_for_view(mesh_info)

    results = []
    with tempfile.TemporaryDirectory(prefix="render_") as tmp_dir:
        for cam_info in cameras:
            view_name = cam_info['name']
            cam = cam_info['camera']
            description = cam_info['description']

            # 渲染图像
            img = renderer.create_view_image(shape, cam, view_name)

            # 保存到临时文件
            tmp_path = os.path.join(tmp_dir, f"view_{view_name}.png")
            img.save(tmp_path)

            # 上传到 OSS
            object_key = f"products/six-views/{int(time.time())}_{view_name}.png"
            url = oss_client.upload_image(img, object_key)

            results.append({
                'url': url,
                'description': description
            })

    return results

def update_task_status(task_id: str, status: str, error_msg: str = ""):
    """更新 Redis 中的任务状态"""
    task_key = f"stp_render_task:{task_id}"
    redis_client.hset(task_key, mapping={
        'status': status,
        'error_message': error_msg,
        'updated_at': time.strftime('%Y-%m-%dT%H:%M:%SZ')
    })

def notify_go_backend(task_id: str, status: str, six_views: list = None, error_msg: str = ""):
    """通知 Go 后端渲染结果"""
    payload = {
        'task_id': task_id,
        'status': status,
        'error_message': error_msg
    }

    if six_views:
        payload['six_views'] = six_views

    try:
        response = requests.post(
            f"{GO_API_BASE_URL}/api/internal/stp-render/complete",
            json=payload,
            timeout=10
        )
        return response.status_code == 200
    except Exception as e:
        print(f"Failed to notify Go backend: {e}")
        return False

def process_task(task_id: str):
    """处理单个渲染任务"""
    print(f"Processing task: {task_id}")

    # 获取任务信息
    task_key = f"stp_render_task:{task_id}"
    task_info = redis_client.hgetall(task_key)

    if not task_info:
        print(f"Task not found: {task_id}")
        return

    stp_file_url = task_info.get('stp_file_url')
    product_id = task_info.get('product_id')

    # 更新状态为 processing
    update_task_status(task_id, 'processing')

    try:
        # 下载 STP 文件
        print(f"Downloading STP file from: {stp_file_url}")
        stp_path = download_stp_file(stp_file_url)

        # 渲染六面图
        print(f"Rendering six views for product {product_id}")
        six_views = render_six_views(stp_path)

        # 通知 Go 后端
        notify_go_backend(task_id, 'completed', six_views)
        update_task_status(task_id, 'completed')

        print(f"Task {task_id} completed successfully")

    except Exception as e:
        error_msg = str(e)
        print(f"Task {task_id} failed: {error_msg}")

        # 重试逻辑
        retry_count = int(task_info.get('retry_count', 0))
        if retry_count < MAX_RETRIES:
            redis_client.hincrby(task_key, 'retry_count', 1)
            # 指数退避
            delay = [10, 30, 60][retry_count]
            time.sleep(delay)
            # 重新入队
            redis_client.rpush(QUEUE_NAME, task_id)
        else:
            notify_go_backend(task_id, 'failed', error_msg=error_msg)
            update_task_status(task_id, 'failed', error_msg)

def worker_loop():
    """Worker 主循环"""
    print("STP Renderer Worker started")
    print(f"Redis: {REDIS_HOST}:{REDIS_PORT}/{REDIS_DB}")
    print(f"Go API: {GO_API_BASE_URL}")

    while True:
        try:
            # 阻塞等待任务（BRPOP）
            result = redis_client.blpop(QUEUE_NAME, timeout=30)
            if result:
                _, task_id = result
                process_task(task_id)
        except KeyboardInterrupt:
            print("Worker stopped")
            break
        except Exception as e:
            print(f"Worker error: {e}")
            time.sleep(5)

if __name__ == "__main__":
    worker_loop()
```
